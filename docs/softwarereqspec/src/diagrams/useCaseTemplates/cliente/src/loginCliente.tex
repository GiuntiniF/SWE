\begin{tabularx}{\textwidth}{|l|X|}
  \hline
  \ucid{A-02} \\ \hline
  \ucname{Login Atleta} \\ \hline
  \ucactor{Atleta Registrato} \\ \hline
  \ucdescription{L'atleta inserisce le proprie credenziali
    di accesso per accedere al sistema.} \\ \hline
  \ucprecond{L'atleta deve aver già completato la
    registrazione all'applicativo.} \\ \hline
  \ucpostcond{L'atleta è autenticato e può usare le
    funzioni dell'applicativo.} \\ \hline
  \ucmainflow{
    \item L'atleta atterra nella pagina di login.
    \item Il sistema mostra il form di inserimento mail e password.
    \item L'atleta inserisce le credenziali.
    \item L'atleta preme il pulsante "Accedi".
    \item Il sistema verifica la mail nel database.
  } \\ \hline
  \ucaltflow{
    \textbf{3a. Password dimenticata} \newline
    - L'atleta riceve una mail con un link per il ripristino della
    password.\newline
    - Una volta reimpostata la password l'atleta è autenticato e può
    usare le funzioni dell'applicativo.\newline
    \textbf{5a. Credenziali errate} \newline
    - Il sistema mostra un messaggio di errore ("Email o password non
    valida"). \newline
    - Il flusso riprende dal punto 2. \newline
  } \\ \hline
\end{tabularx}
